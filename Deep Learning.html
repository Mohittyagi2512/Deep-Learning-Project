for page in range(1, 21):
    url = f"https://www.flipkart.com/search?q=mobiles&page={page}"

    response = requests.get(url, headers=headers)
    soup = BeautifulSoup(response.content, "html.parser")

    items = soup.find_all("div", {"class": "_1AtVbE"})

    for item in items:
        try:
            name = item.find("div", class_="_4rR01T")
            name = name.text if name else None

            price = item.find("div", class_="_30jeq3 _1_WHN1")
            price = price.text if price else None

            rating = item.find("div", class_="_3LWZlK")
            rating = rating.text if rating else None

            products.append({
                "Product_Name": name,
                "Price": price,
                "Rating": rating
            })

        except Exception as e:
            print(e)

    print(f"Page {page} scraped successfully")
    time.sleep(2)
